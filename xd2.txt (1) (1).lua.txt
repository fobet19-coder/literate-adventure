-- ==========================================
-- 1. BUSCADOR Y EQUIPADOR DE PUÑO / ARMA
-- ==========================================
local function buscarYEquiparPuno(character, humanoid)
    local palabrasClave = {"punch", "fist", "puño", "puno", "golpe", "combat", "melee", "bat", "hit"}

    local herramientaEquipada = character:FindFirstChildOfClass("Tool")
    if herramientaEquipada then
        return herramientaEquipada
    end

    local backpack = player:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                local nombreLower = string.lower(item.Name)
                for _, clave in ipairs(palabrasClave) do
                    if string.find(nombreLower, clave) then
                        humanoid:EquipTool(item)
                        return item
                    end
                end
            end
        end

        local primeraTool = backpack:FindFirstChildOfClass("Tool")
        if primeraTool then
            humanoid:EquipTool(primeraTool)
            return primeraTool
        end
    end

    return nil
end

-- ==========================================
-- 2. DETECTOR Y ROMPEDOR DE VENTANAS EN WORKSPACE
-- ==========================================
local function esVentana(objeto)
    if not objeto or not objeto:IsA("BasePart") or not objeto.CanCollide then
        return false
    end

    local nombre = string.lower(objeto.Name)
    local nombrePadre = objeto.Parent and string.lower(objeto.Parent.Name) or ""

    if string.find(nombre, "window") or string.find(nombre, "glass") or string.find(nombre, "ventana") or string.find(nombre, "cristal")
        or string.find(nombrePadre, "window") or string.find(nombrePadre, "glass") or string.find(nombrePadre, "ventana") then
        return true
    end

    if objeto.Material == Enum.Material.Glass or (objeto.Transparency >= 0.15 and objeto.Transparency <= 0.92) then
        return true
    end

    return false
end

local function revisarYRomperVentana(rootPart, character, humanoid, direccion)
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude

    local resultado = Workspace:Raycast(rootPart.Position, direccion * 6.5, rayParams)
    if resultado and resultado.Instance and esVentana(resultado.Instance) then
        local ventana = resultado.Instance

        if tick() - ultimoGolpe >= 0.25 then
            ultimoGolpe = tick()

            local puno = buscarYEquiparPuno(character, humanoid)

            local mirarVentana = Vector3.new(ventana.Position.X, rootPart.Position.Y, ventana.Position.Z)
            if (mirarVentana - rootPart.Position).Magnitude > 0.1 then
                rootPart.CFrame = CFrame.lookAt(rootPart.Position, mirarVentana)
            end

            if aimStateRemote and aimStateRemote:IsA("RemoteEvent") then
                pcall(function()
                    aimStateRemote:FireServer(ventana.Position, true)
                end)
            end

            if puno then
                puno:Activate()
            else
                pcall(function()
                    VirtualInputManager:SendMouseButtonEvent(500, 500, 0, true, game, 1)
                    task.wait(0.04)
                    VirtualInputManager:SendMouseButtonEvent(500, 500, 0, false, game, 1)
                end)
            end
        end
        return true
    end
    return false
end

-- ==========================================
-- 4. FÍSICAS DE SALTO, SUELO Y MOVIMIENTO
-- ==========================================
local function ejecutarSaltoValla(humanoid, rootPart)
    if tick() - ultimoSalto < 0.75 then return end
    ultimoSalto = tick()

    humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    local velActual = rootPart.AssemblyLinearVelocity
    rootPart.AssemblyLinearVelocity = Vector3.new(velActual.X, math.max(humanoid.JumpPower, 55), velActual.Z)
end

local function revisarVallaEnfrente(rootPart, character, direccion, pasoActual)
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude

    local alcance = math.clamp(pasoActual + 3.5, 4, 10)
    local origenBajo = rootPart.Position - Vector3.new(0, 0.8, 0)
    local hitBajo = Workspace:Raycast(origenBajo, direccion * alcance, rayParams)

    if not hitBajo or hitBajo.Normal.Y > 0.5 or esVentana(hitBajo.Instance) then
        return false
    end

    local origenAlto = rootPart.Position + Vector3.new(0, 3.8, 0)
    local hitAlto = Workspace:Raycast(origenAlto, direccion * alcance, rayParams)

    return (hitAlto == nil)
end

local function obtenerAlturaSuelo(posicionXZ, character, humanoid, rootPart)
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude

    local origen = Vector3.new(posicionXZ.X, rootPart.Position.Y + 6, posicionXZ.Z)
    local resultado = Workspace:Raycast(origen, Vector3.new(0, -30, 0), rayParams)

    if resultado then
        local alturaCadera = humanoid.HipHeight + (rootPart.Size.Y * 0.5)
        return resultado.Position.Y + alturaCadera
    end
    return rootPart.Position.Y
end

local function calcularTramo(origenPos, direccion)
    local path = PathfindingService:CreatePath({
        AgentRadius = 2.2,
        AgentHeight = 5,
        AgentCanJump = true,
        AgentCanClimb = true,
        WaypointSpacing = 16
    })

    local destino = origenPos + (direccion * DISTANCIA_TRAMO)
    local exito = pcall(function()
        path:ComputeAsync(origenPos, destino)
    end)

    if exito and path.Status == Enum.PathStatus.Success then
        return path:GetWaypoints()
    end
    return nil
end

local function cicloPrincipal()
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")

    if not humanoid or not rootPart then
        detenerTodo("Deslizar + Romper: OFF")
        return
    end

    humanoid.AutoRotate = false

    while desplazando do
        character = player.Character
        humanoid = character and character:FindFirstChildOfClass("Humanoid")
        rootPart = character and character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not rootPart then
            detenerTodo("Deslizar + Romper: OFF")
            break
        end

        humanoid.AutoRotate = false
        local dirActual = rootPart.CFrame.LookVector
        local waypointsActuales = calcularTramo(rootPart.Position, dirActual)

        if not waypointsActuales then
            local hayVentana = revisarYRomperVentana(rootPart, character, humanoid, dirActual)
            if hayVentana then
                task.wait(0.1)
                continue
            else
                detenerTodo("Camino cerrado (OFF)")
                task.wait(1.2)
                if not desplazando then boton.Text = "Deslizar + Romper: OFF" end
                break
            end
        end

        local siguientesWaypoints = nil
        local preCalculando = false
        local totalPuntos = #waypointsActuales

        for i = 2, totalPuntos do
            if not desplazando then break end

            if not preCalculando and i >= math.floor(totalPuntos * 0.55) then
                preCalculando = true
                local ultimoWp = waypointsActuales[totalPuntos]
                local penultimoWp = waypointsActuales[math.max(1, totalPuntos - 1)]
                local dirFinal = Vector3.new(ultimoWp.Position.X - penultimoWp.Position.X, 0, ultimoWp.Position.Z - penultimoWp.Position.Z)
                dirFinal = dirFinal.Magnitude > 0.1 and dirFinal.Unit or rootPart.CFrame.LookVector

                task.spawn(function()
                    siguientesWaypoints = calcularTramo(ultimoWp.Position, dirFinal)
                end)
            end

            local wp = waypointsActuales[i]
            if wp.Action == Enum.PathWaypointAction.Jump then
                ejecutarSaltoValla(humanoid, rootPart)
            end

            while desplazando and rootPart and rootPart.Parent do
                local deltaTime = RunService.Heartbeat:Wait()
                rootPart.AssemblyAngularVelocity = Vector3.zero

                local posActual = rootPart.Position
                local objetivoHorizontal = Vector3.new(wp.Position.X, posActual.Y, wp.Position.Z)
                local vectorPlano = objetivoHorizontal - posActual
                local distanciaPlana = vectorPlano.Magnitude

                local paso = velocidad * deltaTime

                if distanciaPlana <= math.max(2.5, paso * 1.15) then
                    break
                end

                local dirPlana = vectorPlano.Unit

                local bloqueadoPorVentana = revisarYRomperVentana(rootPart, character, humanoid, dirPlana)
                if bloqueadoPorVentana then
                    continue
                end

                if revisarVallaEnfrente(rootPart, character, dirPlana, paso) then
                    ejecutarSaltoValla(humanoid, rootPart)
                end

                local siguientePosXZ = posActual + (dirPlana * paso)

                local enElAire = (tick() - ultimoSalto < 0.5)
                    or humanoid:GetState() == Enum.HumanoidStateType.Jumping
                    or humanoid:GetState() == Enum.HumanoidStateType.Freefall

                local yFinal
                if enElAire then
                    yFinal = posActual.Y
                else
                    rootPart.AssemblyLinearVelocity = Vector3.new(0, rootPart.AssemblyLinearVelocity.Y, 0)
                    yFinal = obtenerAlturaSuelo(siguientePosXZ, character, humanoid, rootPart)
                end

                local nuevaPosicion = Vector3.new(siguientePosXZ.X, yFinal, siguientePosXZ.Z)
                local rotacionActual = rootPart.CFrame.Rotation
                local rotacionObjetivo = CFrame.lookAt(Vector3.zero, dirPlana).Rotation
                local rotacionSuave = rotacionActual:Lerp(rotacionObjetivo, 0.35)

                rootPart.CFrame = CFrame.new(nuevaPosicion) * rotacionSuave
            end
        end

        if not desplazando then break end
        if siguientesWaypoints then
            waypointsActuales = siguientesWaypoints
        end
    end

    if player.Character and player.Character:FindFirstChildOfClass("Humanoid") then
        player.Character:FindFirstChildOfClass("Humanoid").AutoRotate = true
    end
end
